import CodeFrame from "../../components/landing/CodeFrame.astro";
import Features from "../../components/landing/Features.astro";
import Split from "../../components/landing/Split.astro";

export const policy = `namespace com/example/user_management

shape User {
  role: string
  status: string
}

policy user_access {
  fact user: User as currentUser

  rule allow_admin = default false {
    yield currentUser.role == "admin"
  }

  rule allow_user = default false {
    yield allow_admin or (currentUser.role == "user" and currentUser.status == "active")
  }

  export decision of allow_admin
  export decision of allow_user
}`;

export const execCommand = `sentrie exec com/example/user_management/user_access \\
  --facts '{"user": {"role": "user", "status": "active"}}'`;

export const execOutput = `Namespace: com/example/user_management
Policy:    user_access

Rules:
  ✓ allow_user: ✓ True
  ✓ allow_admin: ⨯ False

Values:
  ✓ allow_user: true
  ✓ allow_admin: false`;

export const request = `curl -X POST http://localhost:7529/decision/com/example/user_management/user_access/allow_user \\
  -H "Content-Type: application/json" \\
  -d '{"facts": {"user": {"role": "user", "status": "active"}}}'`;

export const response = `{
  "decisions": [
    {
      "policy": "user_access",
      "namespace": "com/example/user_management",
      "rule": "allow_user",
      "decision": { "state": "true", "value": true },
      "attachments": {},
      "trace": { ... }
    }
  ]
}`;
